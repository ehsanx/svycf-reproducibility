% ---------------------------------------------------------------- the four terms of the expansion
% One card: #1 colour, #2 symbol (math), #3 heading, #4 text, #5 the result that controls the term.
\newcommand{\ProofCardWidth}{7.75}
\newcommand{\ProofCardHeight}{8.3}
\newcommand{\ProofCardStep}{8.85}
\newcommand{\ProofCard}[5]{\parbox[t][\Dg{\ProofCardHeight}cm][t]{\Dg{\ProofCardWidth}cm}{\raggedright
  {\DgFont{34}{38}\color{#1}$\bm{#2}$}\\[\Dg{1}mm]{\bfseries #3}\par\vspace{\Dg{2}mm}#4\par\vfill\textbf{#5}}}
% #1-#4 the cards of S_des, T_1, T_2 and S_mod; #5 label over the first card, #6 label over the other three
\newcommand{\ProofMap}[6]{%
\begin{tikzpicture}[x=\Dg{1}cm,y=\Dg{1}cm,font=\DgFont{23}{27},
  cardbox/.style={draw=##1,line width=\Dg{2.5}pt,fill=##1!7,rounded corners=\Dg{4}mm,inner sep=\Dg{3.5}mm,
    anchor=north west}]
  \node[cardbox=cblue] (a) at (0,0) {#1};
  \node[cardbox=cgray] (b) at (\ProofCardStep,0) {#2};
  \node[cardbox=cgray] (c) at (\fpeval{2*\ProofCardStep},0) {#3};
  \node[cardbox=cgray] (d) at (\fpeval{3*\ProofCardStep},0) {#4};
  \draw[line width=\Dg{3}pt,cblue] ([yshift=\Dg{4}mm]a.north west) -- ([yshift=\Dg{4}mm]a.north east)
    node[midway,above=\Dg{1}mm,font=\DgFont{22}{26}\bfseries,text=cblue]{#5};
  \draw[line width=\Dg{3}pt,inkmuted] ([yshift=\Dg{4}mm]b.north west) -- ([yshift=\Dg{4}mm]d.north east)
    node[midway,above=\Dg{1}mm,font=\DgFont{22}{26}\bfseries,text=inkmuted]{#6};
\end{tikzpicture}}
